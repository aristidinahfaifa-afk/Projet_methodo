gen_data <- function() {
   matrix(rnorm(n * q, mean = mu , sd = sigma), nrow = n, ncol = q)

# Test de Wald

wald_p <- function(x, g1, g2) {
  d     <- sqrt(sum((colMeans(x[g1, , drop = FALSE]) -
                     colMeans(x[g2, , drop = FALSE]))^2))
  scale <- sigma * sqrt(1 / length(g1) + 1 / length(g2))
  pchisq((d / scale)^2, df = q, lower.tail = FALSE)
}

# QQ-plot (density vs uniform density)

qq_unif <- function(p, main) {
  plot(ppoints(length(p)), sort(p), type = "l", lwd = 2,
       xlim = c(0, 1), ylim = c(0, 1),
       xlab = "Theoretical Quantiles", ylab = "Empirical Quantiles",
       main = main)
  abline(0, 1, lty = 2, col = "grey40")   # droite y = x : p-values uniformes
}

n1 <- length(C1); n2 <- length(C2)

d <- sqrt(sum(ecart^2))                     # ||xbar_C1 - xbar_C2||
u <- ecart / d                              # dir(xbar_C1 - xbar_C2)
cat("d =", round(d, 3), " u =", round(u, 3), "\n")

x_prime <- function(phi) {
  xp <- x
  xp[C1, ] <- t(t(x[C1, , drop = FALSE]) + n2/(n1+n2) * (phi - d) * u)
  xp[C2, ] <- t(t(x[C2, , drop = FALSE]) - n1/(n1+n2) * (phi - d) * u)
  xp
}

est_cluster <- function(clust, G) {           # G est-il EXACTEMENT un cluster de clust ?
  k <- clust[G[1]]
  all(clust[G] == k) && sum(clust == k) == length(G)
}
dans_S <- function(phi) {
